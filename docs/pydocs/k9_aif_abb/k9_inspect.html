k9_aif_abb.k9_inspect

k9_inspect — static conformance inspection of K9-AIF solutions (1.15).

Reads a solution folder (never executes it) and reports, per rule, where it departs from the framework's architecture: ABB contracts, three-layer decoupling, the single LLM call path, governance by construction, Kafka ownership, configuration and secret hygiene, squad YAML structure.

from k9_aif_abb.k9_inspect import K9Inspector
report = K9Inspector().inspect("examples/MyApp")
report.verdict, report.score, report.to_markdown()

CLI: k9aif inspect <folder> [--json] [--fail-on violation]. UI: k9x_inspector (k9x-ecosystem) — local folder or GitHub URL.

117    def files(self) -> Iterable[Path]:
118        for dirpath, dirnames, filenames in os.walk(self.root):
119            dirnames[:] = [d for d in dirnames if d not in SKIP_DIRS and not d.startswith(".")
120                           or d in (".github",)]
121            for fn in filenames:
122                p = Path(dirpath) / fn
123                try:
124                    if p.is_file() and p.stat().st_size <= MAX_FILE_BYTES:
125                        yield p
126                except OSError:
127                    continue

129    def rel(self, p: Path) -> str:
130        return str(p.relative_to(self.root))

132    def _load(self) -> None:
133        for p in self.files():
134            if p.suffix == ".py":
135                src = p.read_text(errors="replace")
136                try:
137                    tree, err = ast.parse(src, filename=str(p)), None
138                except SyntaxError as exc:
139                    tree, err = None, f"line {exc.lineno}: {exc.msg}"
140                mod = PyModule(p, self.rel(p), src, tree, err)
141                if tree is not None:
142                    for node in ast.walk(tree):
143                        if isinstance(node, ast.ClassDef):
144                            mod.classes.append(ClassInfo(node.name, mod, node, [_base_name(b) for b in node.bases]))
145                        elif isinstance(node, (ast.Import, ast.ImportFrom)):
146                            mod.imports.append(node)
147                self.modules.append(mod)
148            elif p.suffix in (".yaml", ".yml"):
149                text = p.read_text(errors="replace")
150                try:
151                    docs = [d for d in yaml.safe_load_all(text) if d is not None]
152                    data, err = (docs[0] if len(docs) == 1 else docs), None
153                except yaml.YAMLError as exc:
154                    data, err = None, str(exc).splitlines()[0]
155                self.yaml_docs.append(YamlDoc(p, self.rel(p), data, err, text))
156            else:
157                self.other_files.append(p)
158
159    def _resolve_roles(self) -> None:
160        local = {c.name: c for m in self.modules for c in m.classes}
161
162        def ancestors(cls: ClassInfo, seen: Set[str]) -> Set[str]:
163            out: Set[str] = set()
164            for b in cls.bases:
165                if b in seen:
166                    continue
167                if b in local and local[b] is not cls:
168                    out |= ancestors(local[b], seen | {b})
169                else:
170                    out.add(b)
171            return out
172
173        for m in self.modules:
174            for c in m.classes:
175                anc = ancestors(c, {c.name})
176                for role, bases in ROLE_BASES.items():
177                    hit = anc & bases
178                    if hit:
179                        c.role, c.framework_bases = role, hit
180                        break

182    # ── queries ───────────────────────────────────────────────────────────────
183    def classes(self, role: Optional[str] = None, include_tests: bool = False) -> List[ClassInfo]:
184        return [c for m in self.modules if include_tests or not m.is_test
185                for c in m.classes if role is None or c.role == role]

187    def code_modules(self) -> List[PyModule]:
