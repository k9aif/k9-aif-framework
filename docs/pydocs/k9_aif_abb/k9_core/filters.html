k9_core.filters: K9X Filters (1.32.0, opt-in with filters.enabled).

Filters shape what flows through an agent: in, on the payload before the agent's code runs; out, on its output. Governance decides, filters transform. See base_filter.py and docs/roadmap/k9x-filters-design.md; OOB filters are in k9_aif_abb.k9_filters.

185class FilterCall:
186    """The filters for one agent call: ``run_in`` before governance pre, ``run_out`` before
187    governance post. Both share ``state``."""
188
189    def __init__(self, chain: FilterChain, agent: Any):
190        self.chain = chain
191        self.agent = agent
192        self.state: Dict[str, Any] = {}
193
194    def _ctx(self, direction: str) -> FilterContext:
195        return FilterContext(agent=type(self.agent).__name__, layer=getattr(self.agent, "layer", ""),
196                             config=getattr(self.agent, "config", {}) or {}, direction=direction,
197                             state=self.state)
198
199    def _run(self, direction: str, value: Any) -> Tuple[Any, Dict[str, Any]]:
200        from k9_aif_abb.k9_utils.trace_events import emit_trace_event
201
202        fields: Dict[str, Any] = {}
203        for f in self.chain.filters:
204            if direction not in f.applies_to:
205                continue
206            label = f.name or type(f).__name__
207            method = f.filter_in if direction == "in" else f.filter_out
208            try:
209                res = method(value, self._ctx(direction))
210            except PermissionError as exc:
211                raise FilterError(f"filter {label} raised PermissionError ({exc}): filters don't block; "
212                                  "a block belongs in a Shield check") from exc
213            if not isinstance(res, FilterResult):
214                raise FilterError(f"filter {label}.filter_{direction}() returned {type(res).__name__}, "
215                                  "not a FilterResult")
216            value = res.value
217            if res.result_fields:
218                if direction != "out":
219                    raise FilterError(f"filter {label}: result_fields are for out filters only")
220                fields.update(res.result_fields)
221            if res.record:
222                event = {"type": "FilterApplied", "agent": type(self.agent).__name__, "filter": label,
223                         "filter_type": f.filter_type, "direction": direction, "record": res.record}
224                log.info("[filters] %s %s %s: %s", type(self.agent).__name__, direction, label, res.record)
225                emit_trace_event(event)
226        return value, fields
227
228    def run_in(self, payload: Any) -> Any:
229        value, _ = self._run("in", payload)
230        return value
231
232    def run_out(self, output: Any) -> Tuple[Any, Dict[str, Any]]:
233        return self._run("out", output)
234
235    def has_out(self) -> bool:
236        return any("out" in f.applies_to for f in self.chain.filters)

134class FilterChain:
135    """The filters one agent runs, in chain order."""
136
137    def __init__(self, filters: List[BaseFilter]):
138        for f in filters:
139            if not isinstance(f, BaseFilter):
140                raise TypeError(f"{f!r} is not a BaseFilter")
141            bad = [d for d in f.applies_to if d not in DIRECTIONS]
142            if bad:
143                raise ValueError(f"filter {f.name or type(f).__name__}: applies_to {bad} (use 'in' and/or 'out')")
144        self.filters = list(filters)
145
146    @classmethod
147    def from_config(cls, config: Dict[str, Any]) -> "FilterChain":
148        """Build the chain from ``filters.chain``, narrowed by ``agent_filters`` when the agent's
149        config has it. An unknown name or a filter that can't be built raises: never a silent skip."""
150        from k9_aif_abb.k9_factories.filter_factory import FilterFactory
151
152        specs = list(((config or {}).get("filters") or {}).get("chain") or [])
153        names = [_spec_name(s) for s in specs]
154        settings = agent_filter_settings(config)
155        wanted = settings.get("use")
156        if wanted is not None:
157            if not isinstance(wanted, (list, tuple)):
158                raise ValueError("agent_filters.use: give a list of filter names ([] for none)")
159            unknown = [w for w in wanted if w not in names]
160            if unknown:
161                raise ValueError(f"agent_filters.use names {unknown}, which are not in filters.chain {names}")
162            specs = [s for s, n in zip(specs, names) if n in wanted]
163        # A filter's settings at the top level of agent YAML would be ignored: say so, once per build.
164        for name in (_spec_name(s) for s in specs):
165            if isinstance((config or {}).get(name), dict) and name not in settings:
166                log.warning("[filters] a top-level %r section is not read by the %s filter: put it under "
167                            "agent_filters.%s", name, name, name)
168        return cls([FilterFactory.create(s) for s in specs])
169
170    @classmethod
171    def for_agent(cls, agent: Any) -> Optional["FilterChain"]:
172        """The agent's chain (built once and kept on the agent), or None when filters are off."""
173        if not filters_enabled(getattr(agent, "config", None)):
174            return None
175        chain = agent.__dict__.get("_k9_filter_chain")
176        if chain is None:
177            chain = cls.from_config(agent.config)
178            agent.__dict__["_k9_filter_chain"] = chain
179        return chain
180
181    def begin(self, agent: Any) -> "FilterCall":
182        return FilterCall(self, agent)

def agent_filter_settings(config: Optional[Dict[str, Any]]) -> Dict[str, Any]: 
79def agent_filter_settings(config: Optional[Dict[str, Any]]) -> Dict[str, Any]:
80    """The agent's ``agent_filters:`` block: ``use:`` (optional list) and one section per filter.
81
82    Filters reserve this one key of agent YAML and nothing else, so no filter can collide with a key
83    another part of the framework (``K9OpenAIAgent``'s ``structured_output:``, ...) or a solution uses.
84    It belongs in agent YAML only: ``AgentLoader`` merges agent YAML over the application config key
85    by key, without merging inside a key."""
86    value = (config or {}).get("agent_filters")
87    if value is None:
88        return {}
89    if not isinstance(value, dict):
90        raise ValueError("agent_filters: give a mapping: use: [names] (optional) and a section per filter, "
91                         "e.g. {use: [attention], attention: {budget_tokens: 4000}}")
92    return value

247    """Run the out filters on a dict result's ``output`` and merge their result fields."""
248    if call is None or not isinstance(result, dict) or "output" not in result:
249        return result
