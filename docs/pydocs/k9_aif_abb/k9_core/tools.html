k9_aif_abb.k9_core.tools

k9_core.tools: the tool gate (1.25.0, opt-in with security.tool_gate.enabled).

Every tool call made inside a governed agent call passes one gate, the BaseToolDispatcher. It checks the call's values against records the model didn't write, applies the solution's own rules, and runs the call, refuses it, or holds it for a person. See base_tool_dispatcher.py.

122    async def propose(self, call: ToolCall, connector: Any = None,
123                      session: Optional[GateSession] = None) -> Any:
124        """Decide one intercepted call; run it, refuse it, or hold it for a person."""
125        cid = call.correlation_id
126        self.audit.record(cid, "intercepted", call.to_dict())
127        await self._screen(call)
128
129        spec = self.registry.lookup(call.tool_name)
130        checks = self._check_records(spec, call.arguments)
131        facts = {
132            "tool": call.tool_name,
133            "arguments": call.arguments,
134            "declared": spec.declared,
135            "irreversible": spec.irreversible,
136            "untrusted_input": session.untrusted_input if session else True,
137            "mismatches": checks["mismatches"],
138            "verification_unavailable": checks["unavailable"],
139            "agent": call.agent_id,
140            "request_id": call.request_id,
141        }
142        verdict, rule, reason = self._decide(spec, facts)
143        self.audit.record(cid, "decided", {"verdict": verdict, "rule": rule, "reason": reason, "facts": facts})
144
145        if verdict == "allow":
146            return await self._run(call, connector, spec, decided_by=rule or "framework")
147        if verdict == "deny":
148            self.audit.record(cid, "denied", {"rule": rule, "reason": reason})
149            raise ToolCallDenied(f"[ToolGate] {call.tool_name} refused: {reason}")
150
151        now = self.clock()
152        record = {
153            "call": call.to_dict(),
154            "digest": call.digest(),
155            "facts": facts,
156            "reason": reason,
157            "rule": rule,
158            "tool_spec": {"name": spec.name, "connector": dict(spec.connector)},
159            "created_at": now,
160            "expires_at": now + self.deadline_seconds,
161        }
162        self.audit.create_pending(cid, record)
163        if connector is not None:
164            self._held_connectors[cid] = connector
165        review = self._review(call, facts, reason, record["expires_at"])
166        exc = ToolCallPendingReview(reason, context=review, correlation_id=cid,
167                                    queue=spec.queue or self.queue)
168        if session is not None:
169            session.hold(exc)
170        self.audit.record(cid, "held_for_review", {"queue": spec.queue or self.queue, "reason": reason})
171        raise exc

180    async def _screen(self, call: ToolCall) -> None:
181        from k9_aif_abb.k9_core.governance.pipeline import NoopGovernance
182        from k9_aif_abb.k9_security.tool_result_guard import screen_tool_call
183        governance = self.governance if self.governance is not None else NoopGovernance()
184        try:  # NoopGovernance is refused outside development/test by screen_tool_call itself

187        except PermissionError as exc:

191    def _check_records(self, spec: ToolSpec, arguments: Dict[str, Any]) -> Dict[str, List[Dict[str, Any]]]:
192        mismatches: List[Dict[str, Any]] = []
193        unavailable: List[Dict[str, Any]] = []
194        for check in spec.verify:
195            where = {"argument": check.argument, "entity": check.entity, "field": check.field}
196            if check.argument not in arguments or check.id_from not in arguments:
197                mismatches.append({**where, "problem": "argument missing from the call"})
198                continue
199            if self.origin is None:
200                unavailable.append({**where, "problem": "no k9x Origin configured to look the record up"})
201                continue
202            try:
203                with framework_tool_call():
204                    answer = self.origin.get_current(check.entity, str(arguments[check.id_from]))
205            except Exception as exc:  # the lookup failed: the value can't be checked, so a person decides
206                log.warning("[ToolGate] record check %s.%s failed: %s", check.entity, check.field, exc)
207                unavailable.append({**where, "problem": f"lookup failed: {exc}"})
208                continue
209            problem = _record_problem(answer, check.field, arguments[check.argument])
210            if problem == "unavailable":
211                unavailable.append({**where, "problem": _why_unavailable(answer)})
212            elif problem:
213                mismatches.append({**where, "problem": problem,
214                                   "in_call": arguments[check.argument],
215                                   "on_record": (answer.get("value") or {}).get(check.field)})
216        return {"mismatches": mismatches, "unavailable": unavailable}

218    def _decide(self, spec: ToolSpec, facts: Dict[str, Any]):
219        verdict, rule, reason = None, None, None
220        if self.rules_adapter is not None:
221            out = self.rules_adapter.execute({"tool_call": facts})
222            if out.get("status") != "success":
223                return "hil", None, f"rules engine unavailable: {out.get('error')}"
224            decision = out.get("decision") or {}
225            if not isinstance(decision, dict):
226                decision = {"decision": decision}
227            verdict = decision.get("decision")
228            rule, reason = decision.get("rule"), decision.get("reason")
229            if verdict not in VERDICTS:
230                return "hil", rule, f"rules returned {verdict!r}, not allow/deny/hil"
231
232        # The rules can make a call stricter, never looser.
233        if facts["mismatches"] or facts["verification_unavailable"]:
234            if verdict == "deny":
235                return "deny", rule, reason or "denied by the solution's rules"
236            what = facts["mismatches"] or facts["verification_unavailable"]
237            return "hil", rule, "values don't match the record: " + "; ".join(
238                f"{m['argument']} ({m['problem']})" for m in what)
239        if verdict in ("deny", "hil"):
240            return verdict, rule, reason or f"{verdict} by the solution's rules"
241        if spec.irreversible:
242            if verdict == "allow" and rule:
243                return "allow", rule, reason or f"allowed by rule {rule}"
244            why = "not declared, so treated as irreversible" if not spec.declared else "irreversible"
245            return "hil", rule, f"{spec.name} is {why} and no rule allows it"
246        return "allow", rule, reason or "reversible"

267    async def _run(self, call: ToolCall, connector: Any, spec: ToolSpec, *, decided_by: str) -> Any:
268        connector = connector if connector is not None else self._connector_for(spec)
269        try:
270            with framework_tool_call():
271                result = connector.call_tool(call.tool_name, call.arguments)
272                if inspect.isawaitable(result):
273                    result = await result
274        except Exception as exc:
275            self.audit.record(call.correlation_id, "failed", {"error": str(exc)[:500], "decided_by": decided_by})
276            raise
277        self.audit.record(call.correlation_id, "executed", {"decided_by": decided_by})
278        return result

280    def _connector_for(self, spec: ToolSpec) -> Any:
281        if self.connector_resolver is not None:
282            return self.connector_resolver(spec)
283        if not spec.connector.get("client"):
284            raise RuntimeError(f"[ToolGate] cannot run {spec.name} after review: declare "

293    async def aresume(self, reply: Dict[str, Any]) -> Dict[str, Any]:
294        """Apply a reviewer's reply: run the stored call once on a valid approval, never otherwise."""
295        cid = (reply or {}).get("correlation_id")
296        pending = self.audit.get_pending(cid) if cid else None
297        if pending is None:
298            log.warning("[ToolGate] reply for unknown tool call %s: nothing runs", cid)
299            return {"status": "not_found", "correlation_id": cid}
300        if pending["status"] != "pending":
301            return {"status": "already_decided", "correlation_id": cid, "state": pending["status"]}
302
303        spec = self.registry.lookup(pending["call"]["tool_name"])
304        decision = ToolCallDecision.parse(
305            reply, pending, now=self.clock(), approve_actions=self.approve_actions,
306            require_digest_echo=self.require_digest_echo, require_assurance=spec.require_assurance)
307        if not decision.approved:
308            if self.audit.transition(cid, ["pending"], decision.outcome, decided_by=decision.approver,
309                                     decided_at=decision.decided_at, decision_reason=decision.reason):
310                self.audit.record(cid, decision.outcome, {"by": decision.approver, "reason": decision.reason})
311            return {"status": "not_executed", "outcome": decision.outcome, "reason": decision.reason,
312                    "correlation_id": cid}
313        return await self._run_approved(pending, decided_by=decision.approver, decided_at=decision.decided_at)

319    async def _run_approved(self, pending: Dict[str, Any], *, decided_by: Optional[str],
320                            decided_at: Optional[str]) -> Dict[str, Any]:
321        cid = pending["correlation_id"]
322        # at most once: claim first; a crash after this never re-runs the call
323        if not self.audit.transition(cid, ["pending"], "executing", decided_by=decided_by, decided_at=decided_at):
324            return {"status": "already_decided", "correlation_id": cid}
325        self.audit.record(cid, "approved", {"by": decided_by, "at": decided_at})
326        call = ToolCall.from_dict(pending["call"])
327        if call.digest() != pending["digest"]:  # stored call altered after review
328            self.audit.transition(cid, ["executing"], "failed", error="stored call does not match its digest")
329            self.audit.record(cid, "failed", {"error": "stored call does not match its digest"})
330            return {"status": "not_executed", "outcome": "malformed", "correlation_id": cid,
331                    "reason": "stored call does not match its digest"}
332        spec = self.registry.lookup(call.tool_name)
333        connector = self._held_connectors.pop(cid, None)
334        try:
335            result = await self._run(call, connector, spec, decided_by=decided_by or "reviewer")
336        except Exception as exc:
337            self.audit.transition(cid, ["executing"], "failed", error=str(exc)[:500])
338            raise
339        self.audit.transition(cid, ["executing"], "executed")
340        return {"status": "executed", "correlation_id": cid, "result": result}

342    def expire_stale(self) -> List[Dict[str, Any]]:
343        """Apply ``on_expiry`` to reviews past their deadline. Run it on a schedule."""
344        out = []
345        for pending in self.audit.pending_before(self.clock()):
346            cid = pending["correlation_id"]
347            if self.on_expiry == "approve":
348                log.warning("[ToolGate] review for %s expired; on_expiry=approve runs it unapproved", cid)
349                from k9_aif_abb.k9_core.agent.base_agent import _resolve_sync
350                out.append(_resolve_sync(self._run_approved(pending, decided_by="on_expiry", decided_at=None)))
351            elif self.audit.transition(cid, ["pending"], "expired", decision_reason="deadline passed"):
352                self.audit.record(cid, "expired", {"reason": "deadline passed"})
353                out.append({"status": "not_executed", "outcome": "expired", "correlation_id": cid})
354        return out
