k9_aif_abb.k9_origin

k9x Origin — what is true right now, and where it came from.

Built into the framework like k9x Shield. An agent makes one call (get_current); Origin answers from live state (streams), reads through to the system of record when needed, verifies before irreversible steps, flags quarantine and conflicts, and screens requests and answers with k9x Shield. It states facts; business rules (K9RulesAgent) decide.

Design: docs/roadmap/k9x-origin-design.md. Configure in config/config.yaml (origin:) and config/origin.yaml. Build with OriginFactory.create(config).

 72class K9Origin(BaseOrigin):
 73    """
 74    The built-in k9x Origin.
 75
 76    Answering ``get_current(entity, id)`` runs, in order: scope → request screening → live state
 77    → read-through to the system of record when the entity is missing → verify against the
 78    system of record when the asking agent is a ``verify_before`` step → flags (quarantine,
 79    in review, conflict; never hidden) → answer screening → trace.
 80
 81    Ingest is the context fabric's (``ContextIngest``); Origin listens to field changes on the
 82    same state and flags conflicts (reversal, late event; disagreement from verify). Conflicts
 83    are facts: ``conflict_listeners`` receive them (an application's Router turns them into the
 84    ``origin.conflict`` domain event), and business rules decide what follows.
 85    """

 87    def __init__(self, state: BaseContextState, config: Optional[OriginConfig] = None,
 88                 request_governance: Any = None, answer_governance: Any = None,
 89                 sources: Optional[OriginSources] = None):
 90        self.state = state
 91        self.config = config or OriginConfig()
 92        self.request_governance = request_governance
 93        self.answer_governance = answer_governance
 94        self.sources = sources or OriginSources(self.config.sources)
 95        self.conflict_listeners: List[Callable[[Conflict], None]] = []
 96        self.call_listeners: List[Callable[[Dict[str, Any]], None]] = []
 97        self._conflicts: Dict[Key, Conflict] = {}
 98        self._reviews: Dict[Key, Dict[str, Any]] = {}
 99        self._lock = threading.Lock()
100        self.rules: List[BaseConflictRule] = []
101        for name, policy in self.config.entities.items():
102            for watch in policy.conflicts:
103                self.rules.append(TransitionRule(name, watch))
104            if policy.conflicts:
105                self.rules.append(LateEventRule(name, [w.field for w in policy.conflicts]))
106        state.change_listeners.append(self._on_change)

109    def get_current(self, entity: str, entity_id: str, agent: Optional[str] = None) -> Answer:
110        t0 = time.time()
111        entity, entity_id = str(entity), str(entity_id)
112        refused = self._refusal("get_current", {"entity": entity, "id": entity_id}, agent)
113        if refused:
114            return self._done("get_current", entity, entity_id, agent, t0, refused)
115
116        view = self.state.get_current(entity, entity_id)
117        kind = "stream"
118        policy = self.config.entities.get(entity)
119        sor = policy.system_of_record if policy else None
120        unavailable = None
121        if not view.get("found") and sor:
122            got = self._read_through(entity, entity_id, sor)
123            if got is True:
124                view, kind = self.state.get_current(entity, entity_id), "system_of_record"
125            elif isinstance(got, str):              # the source failed: not the same as "no such record"
126                unavailable = {"system": sor, "error": got}
127
128        verified = None
129        if view.get("found") and sor and self.config.verifies(agent):
130            verified = self._verify(entity, entity_id, view, policy)
131            kind = "both"
132
133        answer = self._answer(entity, entity_id, view, kind, verified)
134        if unavailable:
135            answer["source_unavailable"] = unavailable
136        return self._done("get_current", entity, entity_id, agent, t0, self._screen_answer("get_current", answer))

140        t0 = time.time()

142        refused = self._refusal("recent_changes", args, agent)
143        if refused:
144            return self._done("recent_changes", entity, entity_id, agent, t0, refused)
145        out = self.state.recent_changes(entity, entity_id, window_seconds)
146        view = self.state.get_current(entity, entity_id)
147        if view.get("history_available") is False:
148            out["history_available"] = False
149        return self._done("recent_changes", entity, entity_id, agent, t0, self._screen_answer("recent_changes", out))

152              minimum: Optional[Dict[str, float]] = None, agent: Optional[str] = None) -> Answer:
153        t0 = time.time()
154        args = {"entity": entity, "filters": filters or {}, "minimum": minimum or {}}
155        refused = self._refusal("query", args, agent)
156        if refused:
157            return self._done("query", entity, None, agent, t0, refused)

162    def resolve_for_squad(self, squad_id: str, payload: Dict[str, Any]) -> Dict[str, Answer]:
163        refs = self.config.squad_refs(squad_id) + list(payload.get("live_context_refs") or [])
164        fill = _Fill({k: v for k, v in payload.items() if isinstance(v, (str, int, float))})
165        live: Dict[str, Answer] = {}
166        for ref in refs:
167            if isinstance(ref, str) and "{" in ref:
168                try:
169                    ref = string.Formatter().vformat(ref, (), fill)
170                except (KeyError, IndexError, ValueError):
171                    continue   # this payload does not name that entity
172            parsed = _ref(ref)
173            if parsed:
174                live[f"{parsed[0]}:{parsed[1]}"] = self.get_current(*parsed, agent=None)
175        return live

192    def _refusal(self, tool: str, args: Dict[str, Any], agent: Optional[str]) -> Optional[Answer]:
193        entity = str(args.get("entity", "")).lower()
194        scope = self.config.allowed_entities(agent)
195        if scope is not None and entity not in scope:
196            who = f"agent {agent}" if agent else "this solution"
197            return {"refused_by_governance": True, "tool": tool,
198                    "reason": f"entity '{entity}' is outside the scope of {who} ({', '.join(sorted(scope))})"}
199        if self.request_governance is not None:
200            payload = {"query": json.dumps(args, default=str), "source": "origin_request", "source_tool": tool}
201            try:
202                _call(self.request_governance.pre_process, payload, {"component": f"origin:{tool}"})
203            except PermissionError as exc:
204                return {"refused_by_governance": True, "tool": tool, "reason": str(exc)}
205        return None

207    def _screen_answer(self, tool: str, answer: Answer) -> Answer:
208        if self.answer_governance is None:
209            return answer
210        from k9_aif_abb.k9_security.tool_result_guard import _run_coro_sync, screen_tool_result, withheld_value
211        try:
212            _run_coro_sync(screen_tool_result(self.answer_governance, tool, answer, {"component": "k9x-origin"}))
213        except PermissionError as exc:
214            return withheld_value(answer, str(exc))
215        return answer

217    def _read_through(self, entity: str, entity_id: str, source_name: str):
218        """True: found and applied. False: the source has no such record. A string: the source failed (its
219        error), reported to the agent as ``source_unavailable`` (1.19.0; before, it read as "not found")."""
220        try:
221            record = self.sources.fetch(source_name, entity, entity_id)
222        except Exception as exc:
223            log.warning("[Origin] read-through %s %s:%s failed: %s", source_name, entity, entity_id, exc)
224            return str(exc)[:200] or type(exc).__name__
225        if not record:
226            return False
227        version = record.get("version") if isinstance(record, dict) else None
228        env = EventEnvelope(event_type=f"{entity}.read_through", source_system=source_name, payload=record,
229                            metadata={"version": f"{source_name}:{version}"} if version is not None else {})
230        source = {"kind": "system_of_record", "system": source_name}
231        self.state._write(ContextUpdate(entity, entity_id, fields=dict(record)), env, source)
232        self.state.mark_history_unavailable((entity, entity_id))
233        return True
234
235    def _verify(self, entity: str, entity_id: str, view: Answer, policy) -> Dict[str, Any]:
236        sor = policy.system_of_record
237        try:
238            record = self.sources.fetch(sor, entity, entity_id) or {}
239        except Exception as exc:
240            return {"against": sor, "agrees": None, "error": str(exc)[:200]}
241        owned = policy.owned_fields or list(record)
242        live = view.get("value") or {}
243        differences = [{"field": f, "live": live.get(f), "system_of_record": record.get(f)}
244                       for f in owned if f in record and live.get(f) != record.get(f)]
245        if differences:
246            self._raise(Conflict(entity, entity_id, "disagreement", differences[0]["field"],
247                                 f"live state and {sor} disagree on {', '.join(d['field'] for d in differences)}",
248                                 differences))
249        return {"against": sor, "agrees": not differences, "differences": differences}
250
251    def _answer(self, entity: str, entity_id: str, view: Answer, kind: str, verified) -> Answer:
252        key = (entity, str(entity_id))
253        with self._lock:
254            conflict = self._conflicts.get(key)
255            review = self._reviews.get(key)
256        if not view.get("found"):
257            return {"entity": entity, "id": str(entity_id), "found": False, "as_of": None,
258                    "flags": {"quarantined": None, "in_review": bool(review), "conflict": None,
259                              "verification_failed": None}}
260        source = dict(view.get("source") or {})
261        source["kind"] = "both" if kind == "both" else source.get("kind") or kind
262        quarantined = view.get("quarantined") or []
263        out = {"entity": entity, "id": str(entity_id), "found": True, "value": view.get("value") or {},
264               "as_of": view.get("as_of"), "age_seconds": view.get("age_seconds"),
265               "version": view.get("version"), "source": source,
266               "flags": {"quarantined": quarantined[-1] if quarantined else None,
267                         "in_review": bool(review), "review": review,
268                         "conflict": conflict.to_dict() if conflict else None,
269                         # a verify_before step whose system of record couldn't be reached (1.19.0): the
270                         # live value is unverified, so it must not be acted on as if checked
271                         "verification_failed": ({"against": verified.get("against"), "error": verified.get("error")}
272                                                 if verified and verified.get("agrees") is None else None)},
273               "verified": verified,
274               "history_available": view.get("history_available", True)}
275        return out

277    def _on_change(self, change: FieldChange) -> None:
278        for rule in self.rules:
279            conflict = rule.check(change)
280            if conflict:
281                self._raise(conflict)
282
283    def _raise(self, conflict: Conflict) -> None:
284        with self._lock:
285            self._conflicts[(conflict.entity, conflict.entity_id)] = conflict
286        emit_trace_event({"type": "OriginConflict", "entity": conflict.entity, "id": conflict.entity_id,
287                          "kind": conflict.kind, "field": conflict.field})
288        for listener in self.conflict_listeners:
289            try:
290                listener(conflict)
291            except Exception:
292                log.exception("[Origin] conflict listener failed")
293
294    def _done(self, tool: str, entity: str, entity_id: Optional[str], agent: Optional[str], t0: float,
295              out: Answer) -> Answer:
296        outcome = ("refused" if out.get("refused_by_governance") else
297                   "withheld" if out.get("withheld_by_governance") else "answered")
298        record = {"tool": tool, "entity": entity, "id": entity_id, "agent": agent,
299                  "ms": round((time.time() - t0) * 1000, 1), "outcome": outcome,
300                  "source": (out.get("source") or {}).get("kind"), "as_of": out.get("as_of"),
301                  "age_seconds": out.get("age_seconds"), "found": out.get("found", True),
302                  "reason": out.get("reason")}
303        emit_trace_event({"type": "Origin", **{k: v for k, v in record.items() if v is not None}})
304        for listener in self.call_listeners:
305            try:
306                listener(record)
307            except Exception:
308                log.exception("[Origin] call listener failed")
309        return out
