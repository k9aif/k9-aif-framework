 26    """
 27    OOB Router State Store SBB
 28    --------------------------
 29    Provides domain-specific persistence for:
 30    - sessions
 31    - session_turns
 32    - routing_decisions
 33    - context_artifacts
 34    - hil_pending (paused flows awaiting a human decision -- see
 35      BaseHILOrchestrator / RequiresHIL)
 36
 37    Supports both:
 38    - PostgreSQL (existing schema / reflection)
 39    - SQLite (OOB auto-created schema)
 40    """

 49    def _init_tables(self):
 50        metadata = self.db.metadata
 51        engine = self.db.engine
 52
 53        try:
 54            # Try reflection first (Postgres / pre-created schema)
 55            self.sessions = Table("sessions", metadata, autoload_with=engine)
 56            self.session_turns = Table("session_turns", metadata, autoload_with=engine)
 57            self.routing_decisions = Table(
 58                "routing_decisions", metadata, autoload_with=engine

 61                "context_artifacts", metadata, autoload_with=engine

 65            # Fallback: create OOB schema (SQLite mode)
 66
 67            self.sessions = Table(
 68                "sessions",
 69                metadata,
 70                Column("session_id", String, primary_key=True),
 71                Column("user_id", String, nullable=False),
 72                Column("status", String, default="ACTIVE"),

 74                Column("model_affinity", String),
 75                Column("active_prefix_hash", Text),
 76                Column("session_summary", Text),
 77                Column("created_at", DateTime),
 78                Column("last_active", DateTime),
 79                Column("closed_at", DateTime),
 80            )

 82            self.session_turns = Table(
 83                "session_turns",
 84                metadata,

 86                Column("session_id", String, nullable=False),
 87                Column("turn_index", Integer, nullable=False),
 88                Column("role", String, nullable=False),
 89                Column("content", Text, nullable=False),
 90                Column("token_count", Integer),

126        # tables above -- an existing DB that predates this table (has

158    def ensure_session(self, session_id: str, user_id: str) -> None:
159        with self.db.get_session() as session:
160            stmt = select(self.sessions.c.session_id).where(
161                self.sessions.c.session_id == session_id
162            )
163            exists = session.execute(stmt).fetchone()
164
165            if not exists:
166                session.execute(
167                    insert(self.sessions).values(
168                        session_id=session_id,
169                        user_id=user_id,
170                        created_at=datetime.utcnow(),
171                        last_active=datetime.utcnow(),
172                        status="ACTIVE",
173                    )
174                )
175            else:
176                session.execute(
177                    update(self.sessions)
178                    .where(self.sessions.c.session_id == session_id)
179                    .values(last_active=datetime.utcnow())
180                )
181
182            session.commit()

187                self.sessions.c.session_id == session_id
188            )
189            row = session.execute(stmt).fetchone()
190            return dict(row._mapping) if row else None

222    def append_turn(
223        self,
224        session_id: str,
225        role: str,
226        content: str,
227        token_count: Optional[int] = None,
228        compressed_flag: bool = False,
229    ) -> int:
230        with self.db.get_session() as session:
231            stmt = (
232                select(self.session_turns.c.turn_index)
233                .where(self.session_turns.c.session_id == session_id)
234                .order_by(self.session_turns.c.turn_index.desc())
235            )
236
237            last = session.execute(stmt).fetchone()
238            next_index = (last[0] + 1) if last else 1
239
240            result = session.execute(
241                insert(self.session_turns).values(
242                    session_id=session_id,
243                    turn_index=next_index,
244                    role=role,
245                    content=content,
246                    token_count=token_count,
247                    compressed_flag=compressed_flag,
248                    created_at=datetime.utcnow(),
249                )
250            )
251
252            session.commit()
253
254            inserted_turn_id = result.inserted_primary_key[0]
255            return inserted_turn_id

356    def resolve_hil_pending(self, correlation_id: str) -> None:
357        """Mark a pending HIL flow resolved once the Router has re-dispatched
358        it. Does not delete the row -- kept for audit/traceability, same
359        append-don't-erase posture as the rest of this store."""
360        with self.db.get_session() as session:
361            session.execute(
362                update(self.hil_pending)
363                .where(self.hil_pending.c.correlation_id == correlation_id)
364                .values(status="resolved", resolved_at=datetime.utcnow())
365            )
366            session.commit()
