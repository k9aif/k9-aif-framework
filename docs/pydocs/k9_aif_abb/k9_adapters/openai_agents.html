k9_aif_abb.k9_adapters.openai_agents

OpenAI Agents SDK adapter package for K9-AIF (1.27.0).

Runs OpenAI agents, declared in K9-AIF agent YAML, under K9-AIF governance: Zero Trust and Shield ingress before the run, Zero Trust for every agent that takes a turn, Shield on every tool call, tool result and model reply, and the K9 Model Router choosing the model for every model call. Needs pip install "k9-aif[openai-agents]".

251    def _validate(self, max_handoffs: int) -> None:
252        if self._entry_name not in self._specs:
253            raise ValueError(f"[OpenAIAgentsOrchestratorAdapter] entry agent '{self._entry_name}' has no spec. "
254                             f"Known agents: {sorted(self._specs)}")
255        for name, spec in self._specs.items():
256            unknown_tools = [t for t in spec.get("tools") or [] if t not in self._capabilities]
257            if unknown_tools:
258                raise ValueError(
259                    f"[OpenAIAgentsOrchestratorAdapter] agent '{name}' names tool(s) the adapter never "
260                    f"registered: {unknown_tools}. Registered: {sorted(self._capabilities)}. An agent can only "
261                    "be given registered capabilities; hosted OpenAI tools can't be given at all.")
262            unknown_agents = [h for h in spec.get("handoffs") or [] if h not in self._specs]
263            if unknown_agents:
264                raise ValueError(f"[OpenAIAgentsOrchestratorAdapter] agent '{name}' hands off to unknown "
265                                 f"agent(s): {unknown_agents}")
266        reachable, frontier = set(), [self._entry_name]
267        while frontier:
268            for target in self._specs[frontier.pop()].get("handoffs") or []:
269                if target != self._entry_name and target not in reachable:
270                    reachable.add(target)
271                    frontier.append(target)
272        if len(reachable) > max_handoffs:
273            raise ValueError(
274                f"[OpenAIAgentsOrchestratorAdapter] a run can reach {len(reachable)} agents by handoff "
275                f"({sorted(reachable)}), more than max_handoffs={max_handoffs}. A known multi-step process "
276                "belongs in a K9 squad, where its flow is declared; keep handoff groups small.")

368    async def _execute_flow_governed(self, payload: Dict[str, Any]) -> Dict[str, Any]:
369        with governed_call(self), trace_scope(trace_id_from(payload)):
370            sdk_input = self.adapt_input(payload)
371
372            zt = self.apply_zero_trust(sdk_input)
373            if not zt["allowed"]:
374                raise PermissionError(f"[OpenAIAgentsOrchestratorAdapter] Zero Trust denied run: {zt.get('reason')}")
375
376            governed = await self.apply_pre_governance(sdk_input, self._governance_context())
377            emit_trace_event({"type": "OpenAIRunCheck", "agent": self.adapter_name, "gate": "ingress",
378                              "allowed": True})
379            prompt = (governed or sdk_input).get("prompt") or ""
380            identity = {k: sdk_input[k] for k in _IDENTITY_KEYS if k in sdk_input}
381            hooks = _K9RunHooks(self, identity)
382
383            self.publish_status("OpenAIAgentsRunStarted", {"adapter": self.adapter_name,
384                                                           "entry_agent": self._entry_name})
385            run_result = await self._run(prompt, hooks)
386
387            final = run_result.final_output
388            await self.apply_post_governance({"output": final if isinstance(final, str) else repr(final)},
389                                             self._governance_context())
390            emit_trace_event({"type": "OpenAIRunCheck", "agent": self.adapter_name, "gate": "egress",
391                              "allowed": True})
392            self.publish_status("OpenAIAgentsRunCompleted", {
393                "adapter": self.adapter_name, "last_agent": getattr(run_result.last_agent, "name", None),
394                "model_calls": len(run_result.raw_responses)})
395            return self.adapt_output({"run_result": run_result, "trace": hooks.trace})

397    def _run_config(self) -> RunConfig:
398        export = bool(((self.config or {}).get("openai_agents") or {}).get("export_traces"))
399        kwargs: Dict[str, Any] = {"tracing_disabled": not export, "workflow_name": self.adapter_name}
400        if self.model_routing == "k9":
401            tools_by_alias: Dict[str, bool] = {}
402            for spec in self._specs.values():
403                alias = spec.get("model") or "general"
404                tools_by_alias[alias] = tools_by_alias.get(alias, False) or bool(spec.get("tools"))
405            kwargs["model_provider"] = K9ModelProvider(
406                self.config, tools_by_alias=tools_by_alias, sensitivity=self.sensitivity,
407                required_capabilities=self.required_capabilities, agent_name=self.adapter_name)
408        elif self._direct_provider is not None:
409            kwargs["model_provider"] = self._direct_provider
410        return RunConfig(**kwargs)

412    async def _run(self, prompt: str, hooks: _K9RunHooks):
413        errors: List[BaseException] = []
414        token = _TOOL_ERRORS.set(errors)
415        try:
416            return await self._run_gated(prompt, hooks)
417        except Exception:
418            if errors:                 # the handler's own exception, not the SDK's wrapper
419                raise errors[0]
420            raise
421        finally:
422            _TOOL_ERRORS.reset(token)
423
424    async def _run_gated(self, prompt: str, hooks: _K9RunHooks):
425        entry = self._agents[self._entry_name]
426        if not gate_enabled(self.config):
427            return await Runner.run(entry, prompt, run_config=self._run_config(), hooks=hooks,
428                                    max_turns=self.max_turns)
429        session = GateSession(BaseToolDispatcher.from_config(self.config, governance=self.governance),
430                              agent_id=self.adapter_name)
431        with gate_session(session):
432            result = await Runner.run(entry, prompt, run_config=self._run_config(), hooks=hooks,
433                                      max_turns=self.max_turns)
434        if session.pending:            # a review the SDK swallowed is raised again
435            raise session.pending[0]
436        return result

101def load_agent_specs(agents_dir: str | Path) -> Dict[str, Dict[str, Any]]:
102    """Read agent YAML files (K9-AIF agent format) keyed by ``name``.
103
104    ``AgentLoader`` indexes by ``class:``, which these agents don't have: an OpenAI agent is
105    built by the adapter, not imported. Files without ``name`` are refused, not skipped.
106    """
107    specs: Dict[str, Dict[str, Any]] = {}
108    for path in sorted(Path(agents_dir).glob("*.yaml")):
109        data = yaml.safe_load(path.read_text(encoding="utf-8")) or {}
110        name = (data.get("name") or "").strip()
111        if not name:
112            raise ValueError(f"[OpenAIAgentsOrchestratorAdapter] agent YAML {path.name} has no 'name'")
113        if name in specs:
114            raise ValueError(f"[OpenAIAgentsOrchestratorAdapter] agent '{name}' is defined twice ({path.name})")
115        specs[name] = data
116    if not specs:
117        raise ValueError(f"[OpenAIAgentsOrchestratorAdapter] no agent YAML in {agents_dir}")
