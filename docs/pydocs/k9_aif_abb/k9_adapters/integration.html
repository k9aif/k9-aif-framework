23class BaseIntegrationAdapter(ABC):
24    """
25    Abstract base for all K9-AIF integration adapters.
26
27    Subclasses must implement :meth:`execute`. Each specific adapter ABB
28    (BaseApiAdapter, BaseMessagingAdapter, etc.) provides a template
29    implementation of execute() that delegates to a type-specific abstract
30    method, so concrete SBBs only implement that one method.
31    """
32
33    def __init__(self, config: Optional[Dict[str, Any]] = None) -> None:
34        self.config: Dict[str, Any] = config or {}
35        self.adapter_name: str = self.__class__.__name__
36
37    @abstractmethod
38    def execute(self, payload: Dict[str, Any]) -> Dict[str, Any]:
39        """Invoke the external system and return a K9-AIF-normalised dict."""
40
41    def validate_input(self, payload: Dict[str, Any]) -> None:
42        """Override to enforce required fields. Raise ValueError on invalid input."""
43
44    def handle_error(self, exc: Exception, payload: Dict[str, Any]) -> Dict[str, Any]:
45        """Default error handler — returns structured error dict, never raises."""
46        return {
47            "adapter": self.adapter_name,
48            "status": "error",
49            "error": str(exc),
50            "error_type": type(exc).__name__,
51        }

23class BaseApiAdapter(BaseIntegrationAdapter):
24    """ABB for deterministic HTTP/API calls — no LLM inference."""

26    @abstractmethod
27    def call_endpoint(
28        self,
29        url: str,
30        method: str,
31        headers: Dict[str, str],
32        body: Dict[str, Any],
33    ) -> Any:
34        """Execute the HTTP call. Return raw response (SA decides the shape)."""
35
36    def execute(self, payload: Dict[str, Any]) -> Dict[str, Any]:
37        self.validate_input(payload)
38        configured = self.config.get("url")
39        url     = configured or payload.get("url", "")
40        method  = self.config.get("method", "POST").upper()
41        headers = {**self.config.get("headers", {}), **payload.get("headers", {})}
42        if not configured:
43            # The agent chose this URL: it may not reach internal or metadata addresses unless the
44            # architect allowed it (allowed_hosts / allow_private_networks).
45            check_destination(url, allowed_hosts=self.config.get("allowed_hosts"),
46                              private_ok=bool(self.config.get("allow_private_networks")))
47        try:
48            result = self.call_endpoint(url, method, headers, payload)
49            return {"adapter": self.adapter_name, "status": "success", "result": result}
50        except OutboundRefused:
51            raise                       # a security refusal is a block signal, never an error result
52        except Exception as exc:
53            return self.handle_error(exc, payload)
54
55    def destination_private_ok(self, template: str) -> bool:
56        """Internal addresses are fine when the architect wrote the host into the configured URL (not
57        filled from the payload) or allowed them explicitly."""
58        return bool(self.config.get("allow_private_networks")) or (
59            bool(self.config.get("url")) and not host_is_templated(template))

42    @staticmethod

44        """Register a concrete adapter SBB under a name."""
45        with IntegrationAdapterFactory._lock:
46            IntegrationAdapterFactory._registry[name.lower()] = cls
47
48    @staticmethod

50        """Return an instance of the named adapter. Raises ValueError if unknown."""
51        key = _ADAPTER_TYPE_MAP.get(name.lower(), name.lower())
52        cls = IntegrationAdapterFactory._registry.get(key)
53        if cls is None:
54            available = list(IntegrationAdapterFactory._registry.keys())
55            raise ValueError(
56                f"No adapter registered for '{name}'. "
57                f"Register an SBB first. Available: {available}"
58            )
59        return cls(config=config or {})

64        return IntegrationAdapterFactory.get(adapter_type, config=config)

16    """
17    REST/JSON over HTTP(S) with ``requests``. Config keys: ``url`` (may contain ``{name}``
18    placeholders filled from the payload, e.g. ``https://erp/api/invoices/{id}``), ``method``
19    (default POST; GET sends the payload as query parameters), ``headers``, ``timeout``
20    (default 30). Credentials belong in headers built from environment variables, never in
21    config.yaml.
22
23    Outbound guard (1.18.5): payload values are URL-encoded into the placeholders, the destination is
24    checked before the request and again on every redirect (``allowed_hosts``, ``allow_private_networks``;
25    see outbound_guard.py).
26
27    Registered as ``http`` in IntegrationAdapterFactory.
28    """
29
30    MAX_REDIRECTS = 5
31
32    def call_endpoint(self, url: str, method: str, headers: Dict[str, str], body: Dict[str, Any]) -> Any:
33        import requests
34        from urllib.parse import urljoin, urlsplit
35        fields = {k: v for k, v in body.items() if isinstance(v, (str, int, float))}
36        target = fill_url(url, fields)               # placeholder values URL-encoded: never URL structure
37        allowed = self.config.get("allowed_hosts")
38        private_ok = self.destination_private_ok(url)
39        timeout = float(self.config.get("timeout", 30))
40        for _ in range(self.MAX_REDIRECTS + 1):
41            check_destination(target, allowed_hosts=allowed, private_ok=private_ok)
42            if method == "GET":
43                resp = requests.get(target, headers=headers, params=body.get("params"), timeout=timeout,
44                                    allow_redirects=False)
45            else:
46                resp = requests.request(method, target, headers=headers, json=body.get("json", body),
47                                        timeout=timeout, allow_redirects=False)
48            if resp.status_code in (301, 302, 303, 307, 308) and resp.headers.get("Location"):
49                nxt = urljoin(target, resp.headers["Location"])      # every hop is checked again
50                # The configured host's trust survives a redirect only on that same host.
51                if urlsplit(nxt).hostname != urlsplit(target).hostname:
52                    private_ok = bool(self.config.get("allow_private_networks"))
53                target = nxt
54                if resp.status_code == 303:
55                    method = "GET"
56                continue
57            break
58        else:
59            raise RuntimeError(f"too many redirects (>{self.MAX_REDIRECTS})")
60        if resp.status_code == 404:
61            return None
62        resp.raise_for_status()
63        try:
64            return resp.json()
65        except ValueError:
66            return resp.text
