 22KEY = rsa.generate_private_key(public_exponent=65537, key_size=2048)

 25def _token(iss, **claims):
 26    now = int(time.time())
 27    return jwt.encode({"iss": iss, "iat": now, "exp": now + 300, **claims}, KEY, algorithm="RS256")

 30def _pin(auth):
 31    auth._key_resolver = lambda t: KEY.public_key()
 32    return auth

 45def test_providers_list_builds_in_order_and_chains(monkeypatch):
 46    monkeypatch.setenv("K9_KEY_BOT", "k-1")
 47    auth = AuthenticatorFactory.create({"security": {"auth": {"providers": [
 48        {"type": "keycloak", "base_url": "http://kc:8113", "realm": "k9x", "client_id": "k9x-agents"},
 49        {"type": "api_key", "api_keys": {"bot": {"env": "K9_KEY_BOT", "roles": ["r"]}}},
 50    ]}}})
 51    assert isinstance(auth, ChainedAuthenticator)
 52    kc, keys = auth.authenticators
 53    assert isinstance(kc, KeycloakAuthenticator) and kc.issuer == "http://kc:8113/realms/k9x"
 54    assert isinstance(keys, ApiKeyAuthenticator)
 55    assert auth.authenticate({"api_key": "k-1"}).principal_id == "bot"

 73def test_solution_sbbs_can_register_their_own():
 74    class CACAuthenticator(BaseAuthenticator):
 75        def __init__(self, allowed):
 76            self.allowed = allowed
 77
 78        @classmethod
 79        def from_config(cls, cfg):
 80            return cls(cfg["allowed"])
 81
 82        def authenticate(self, credentials):
 83            dn = credentials.get("client_cert_dn")
 84            return IdentityContext(principal_id=dn, principal_type="user", roles=["analyst"]) \
 85                if dn in self.allowed else None
 86
 87    AuthenticatorFactory.register("cac", CACAuthenticator)
 88    auth = AuthenticatorFactory.create({"security": {"auth": {"providers": [
 89        {"type": "cac", "allowed": ["CN=DOE.JANE.1234567890"]}]}}})
 90    assert auth.authenticate({"client_cert_dn": "CN=DOE.JANE.1234567890"}).roles == ["analyst"]
 91    assert auth.authenticate({"client_cert_dn": "CN=MALLORY"}) is None

103def test_keycloak_reads_realm_and_own_client_roles_only():
104    auth = _pin(KeycloakAuthenticator(base_url="http://kc:8113", realm="k9x", client_id="k9x-agents"))
105    who = auth.authenticate({"bearer_token": _token(
106        KC_ISS, preferred_username="k9aif",
107        realm_access={"roles": ["claims_reader"]},
108        resource_access={"k9x-agents": {"roles": ["claims_auditor"]},
109                         "some-other-app": {"roles": ["admin"]}})})
110    assert who.roles == ["claims_reader", "claims_auditor"]          # never the other client's "admin"
111    assert who.principal_type == "user"
112

114def test_keycloak_service_account_is_an_agent_and_issuer_is_enforced():
115    auth = _pin(KeycloakAuthenticator(issuer=KC_ISS))
116    who = auth.authenticate({"bearer_token": _token(KC_ISS, preferred_username="service-account-claims-bot")})
117    assert who.principal_id == "claims-bot" and who.principal_type == "agent"
118    assert auth.authenticate({"bearer_token": _token("http://evil/realms/k9x", preferred_username="x")}) is None

126# ── Entra ID ─────────────────────────────────────────────────────────────────
127def test_entra_id_app_roles_and_app_only_tokens():
128    auth = _pin(EntraIDAuthenticator(tenant_id="t-1", audience="api://k9x"))
129    assert auth.issuer == "https://login.microsoftonline.com/t-1/v2.0"
130    iss = auth.issuer
131    user = auth.authenticate({"bearer_token": _token(iss, aud="api://k9x", preferred_username="jane@agency.gov",
132                                                     roles=["Claims.Approve"], tid="t-1")})
133    assert user.principal_id == "jane@agency.gov" and user.roles == ["Claims.Approve"] and user.tenant_id == "t-1"
134    app = auth.authenticate({"bearer_token": _token(iss, aud="api://k9x", idtyp="app", azp="claims-bot-app")})
135    assert app.principal_type == "agent" and app.principal_id == "claims-bot-app"
136    assert auth.authenticate({"bearer_token": _token(iss, aud="api://other", preferred_username="x")}) is None

139# ── Okta ─────────────────────────────────────────────────────────────────────
140def test_okta_groups_and_client_credentials():
141    auth = _pin(OktaAuthenticator(domain="agency.okta.com"))
142    iss = "https://agency.okta.com/oauth2/default"
143    assert auth.issuer == iss and auth.jwks_uri == iss + "/v1/keys"
144    user = auth.authenticate({"bearer_token": _token(iss, aud="api://default", sub="jane@agency.gov",
145                                                     groups=["claims_reader"])})
146    assert user.roles == ["claims_reader"] and user.principal_type == "user"
147    bot = auth.authenticate({"bearer_token": _token(iss, aud="api://default", sub="0oa1bot", cid="0oa1bot")})
148    assert bot.principal_type == "agent"

152class _Doc:
153    def __init__(self, doc):
154        self._b = json.dumps(doc).encode()
155
156    def read(self):
157        return self._b
158
159    def __enter__(self):
160        return self
161
162    def __exit__(self, *a):
163        return False

166def test_discovery_reads_jwks_uri_and_refuses_a_mismatched_issuer(monkeypatch):
167    import urllib.request
168    iss = "https://idp.example/tenant"
169    monkeypatch.setattr(urllib.request, "urlopen",
170                        lambda url, timeout=10: _Doc({"issuer": iss, "jwks_uri": iss + "/keys"}))
171    auth = OIDCAuthenticator(issuer=iss, discovery=True)
172    assert auth.jwks_uri == "" and auth._discover_jwks_uri() == iss + "/keys"
173    monkeypatch.setattr(urllib.request, "urlopen",
174                        lambda url, timeout=10: _Doc({"issuer": "https://evil.example", "jwks_uri": "x"}))
175    with pytest.raises(ValueError, match="does not match"):
176        auth._discover_jwks_uri()
