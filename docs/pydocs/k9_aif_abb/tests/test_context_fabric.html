 39class InvoiceProjection(BaseContextProjection):
 40    entity, id_field = "invoice", "invoice_id"
 41
 42    def accepts(self, env):
 43        return env.event_type.startswith("invoice.")

 49class PaymentProjection(BaseContextProjection):
 50    def accepts(self, env):
 51        return env.event_type == "payment.posted"
 52
 53    def project(self, env):
 54        return {}
 55
 56    def updates(self, env):
 57        p = env.payload
 58        return [ContextUpdate("invoice", p["invoice_id"], fields={"status": "paid"},
 59                              append={"payments": {"payment_id": p["payment_id"], "amount": p["amount"]}}),
 60                ContextUpdate("payment", p["payment_id"], fields={"invoice_id": p["invoice_id"]})]

165def test_late_event_never_overwrites_newer_state_but_is_recorded():
166    st, seen = make_state(), []
167    st.change_listeners.append(seen.append)
168    st.apply(_at(env("invoice.created", invoice_id="INV-1", status="approved"), 30))
169    st.apply(_at(env("invoice.updated", invoice_id="INV-1", status="disapproved"), 5))   # newer
170    st.apply(_at(env("invoice.updated", invoice_id="INV-1", status="approved"), 20))     # older, late
171    cur = st.get_current("invoice", "INV-1")
172    assert cur["value"]["status"] == "disapproved", "a late event must not win"
173    assert cur["age_seconds"] >= 4, "as_of is the source time, not the arrival time"
174    types = [c["type"] for c in st.recent_changes("invoice", "INV-1")["changes"]]
175    assert types == ["invoice.created", "late:invoice.updated", "invoice.updated"], "history in source order"
176    late = [c for c in seen if c.late]
177    assert len(late) == 1 and late[0].new == "approved" and late[0].old == "disapproved"

215def test_ingest_end_to_end_with_quarantine():
216    fabric = InMemoryEventFabric()
217    st, window, seen = make_state(), InMemoryContextWindow(), []
218    ingest = ContextIngest(fabric, st, gate=GovernanceEventGate(ShieldGovernance(SHIELD)), window=window,
219                           listeners=[seen.append])
220    ingest.attach(["ap.invoice", "ap.vendor"])
221
222    fabric.publish(env("invoice.created", invoice_id="INV-1", vendor_id="V-1", amount=900, status="approved"),
223                   "ap.invoice")
224    fabric.publish(env("vendor.note", vendor_id="V-1", note=POISON), "ap.vendor")
225
226    assert st.get_current("invoice", "INV-1")["value"]["status"] == "approved"
227    vendor = st.get_current("vendor", "V-1")
228    assert "notes" not in vendor["value"], "a refused event's content is never applied"
229    assert vendor["quarantined"][0]["blocked_by"] in ("PromptInjectionCheck", "PaymentDiversionCheck")
230    assert [e["decision"] for e in seen] == ["PERMIT", "BLOCK"]
231    assert seen[1]["topic"] == "ap.vendor" and seen[1]["entities"] == ["vendor:V-1"]
232    assert window.size() == 1 and ingest.counts == {"permitted": 1, "blocked": 1}

ABB contract for transforming raw enterprise events into agent-usable context.

A projection is a declarative mapping: source-system schema → agent vocabulary. Agents speak domain terms, not SAP IDoc or Salesforce API formats.

Multiple projections can be registered per stream. The first projection whose accepts() returns True is applied.

Example: SAP event: {"VBELN": "P001", "KBETR": 5000.00, "WAERS": "USD"} Projected: {"policy_id": "P001", "premium": 5000.00, "currency": "USD"}

SBBs implement one projection per source-system event type.

Live state (1.17): updates() turns the projection into ContextUpdates for BaseContextState. By default it keys project()'s result by the class attributes entity and id_field::

class PolicyProjection(BaseContextProjection):

Multiple projections can be registered per stream. The first projection whose accepts() returns True is applied.

Example: SAP event: {"VBELN": "P001", "KBETR": 5000.00, "WAERS": "USD"} Projected: {"policy_id": "P001", "premium": 5000.00, "currency": "USD"}

SBBs implement one projection per source-system event type.

Live state (1.17): updates() turns the projection into ContextUpdates for BaseContextState. By default it keys project()'s result by the class attributes entity and id_field::
