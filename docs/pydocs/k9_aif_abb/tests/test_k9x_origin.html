 55def ev(event_type, seconds_ago=0, **payload):

254def test_http_api_adapter_fills_the_url_and_treats_404_as_not_found(monkeypatch):
255    import requests
256
257    class Resp:
258        def __init__(self, code, body):
259            self.status_code, self._body = code, body
260
261        def raise_for_status(self):
262            if self.status_code >= 400:
263                raise requests.HTTPError(str(self.status_code))
264
265        def json(self):
266            return self._body
267
268    calls = []
269    monkeypatch.setattr(requests, "get", lambda url, **kw: calls.append(url) or
270                        (Resp(200, {"status": "paid"}) if url.endswith("/P-1") else Resp(404, None)))
271    adapter = HttpApiAdapter({"url": "https://treasury.example/payments/{id}", "method": "GET"})
272    assert adapter.execute({"id": "P-1"})["result"] == {"status": "paid"}
273    assert adapter.execute({"id": "P-2"})["result"] is None
274    assert calls == ["https://treasury.example/payments/P-1", "https://treasury.example/payments/P-2"]

ABB contract for transforming raw enterprise events into agent-usable context.

A projection is a declarative mapping: source-system schema → agent vocabulary. Agents speak domain terms, not SAP IDoc or Salesforce API formats.

Multiple projections can be registered per stream. The first projection whose accepts() returns True is applied.

Example: SAP event: {"VBELN": "P001", "KBETR": 5000.00, "WAERS": "USD"} Projected: {"policy_id": "P001", "premium": 5000.00, "currency": "USD"}

SBBs implement one projection per source-system event type.

Live state (1.17): updates() turns the projection into ContextUpdates for BaseContextState. By default it keys project()'s result by the class attributes entity and id_field::

class PolicyProjection(BaseContextProjection):
    entity, id_field = "policy", "policy_id"

Override updates() when one event changes several entities, or appends to a list (a payment posted changes the invoice and creates a payment).

def test_http_api_adapter_fills_the_url_and_treats_404_as_not_found(monkeypatch): 
255def test_http_api_adapter_fills_the_url_and_treats_404_as_not_found(monkeypatch):
256    import requests
257
258    class Resp:
259        def __init__(self, code, body):
260            self.status_code, self._body = code, body
261
262        def raise_for_status(self):
263            if self.status_code >= 400:
264                raise requests.HTTPError(str(self.status_code))
265
266        def json(self):
267            return self._body
268
269    calls = []
270    monkeypatch.setattr(requests, "get", lambda url, **kw: calls.append(url) or
271                        (Resp(200, {"status": "paid"}) if url.endswith("/P-1") else Resp(404, None)))
272    adapter = HttpApiAdapter({"url": "https://treasury.example/payments/{id}", "method": "GET"})
273    assert adapter.execute({"id": "P-1"})["result"] == {"status": "paid"}
274    assert adapter.execute({"id": "P-2"})["result"] is None
275    assert calls == ["https://treasury.example/payments/P-1", "https://treasury.example/payments/P-2"]
