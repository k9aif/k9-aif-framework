 47def test_intent_orchestrator_module_never_names_an_agent():
 48    """No import of, and no reference to, an agent class in the orchestrator's code (docstrings may mention it)."""
 49    import ast, inspect
 50    from k9_aif_abb.k9_orchestrators import intent_orchestrator
 51    tree = ast.parse(inspect.getsource(intent_orchestrator))
 52    names = {n.id for n in ast.walk(tree) if isinstance(n, ast.Name)}

 59def test_monitor_factory_uses_the_application_config(tmp_path, monkeypatch):

 82# ── llm_invoke: short waits when the model server is down ───────────────

 84def _invoke_with(outputs_or_errors, **kw):
 85    from unittest.mock import MagicMock, patch
 86    import k9_aif_abb.k9_utils.llm_invoke as mod
 87    from k9_aif_abb.k9_inference.models.inference_request import InferenceRequest
 88    seq = []
 89    for x in outputs_or_errors:
 90        if isinstance(x, Exception):
 91            seq.append(x)
 92        else:
 93            r = MagicMock(); r.output = x; r.model_alias = "m"; r.provider = "p"; r.latency_ms = 1; r.token_usage = {}
 94            seq.append(r)
 95    with patch.object(mod, "ModelRouterFactory") as f, patch.object(mod.time, "sleep") as sleep:
 96        f.get_router.return_value.invoke.side_effect = seq
 97        try:
 98            mod.llm_invoke(kw.pop("config", {}), InferenceRequest(prompt="x", task_type="general"), **kw)
 99        except RuntimeError:
100            pass
101        return [c.args[0] for c in sleep.call_args_list]

104def test_unreachable_model_waits_briefly():
105    assert _invoke_with([ConnectionRefusedError("refused")] * 3) == [5.0, 5.0]
106    assert _invoke_with(["[WARN] connection refused"] * 3) == [5.0, 5.0]       # OllamaLLM's way of saying it

113def test_retry_settings_come_from_config_and_explicit_args_win():
114    cfg = {"inference": {"retry": {"attempts": 2, "delay_seconds": 30, "unreachable_delay_seconds": 1}}}
115    assert _invoke_with(["", ""], config=cfg) == [30.0]
116    assert _invoke_with([ConnectionError("x")] * 2, config=cfg) == [1.0]
117    assert _invoke_with(["", "", ""], config=cfg, max_retries=3, retry_delay_s=10) == [10.0, 10.0]

122def test_trace_resolves_the_model_through_llm_ref():
123    from unittest.mock import MagicMock, patch
124    import k9_aif_abb.k9_utils.llm_invoke as mod
125    from k9_aif_abb.k9_inference.models.inference_request import InferenceRequest
126    cfg = {"inference": {"model_catalog": {"models": {"secure": {"llm_ref": "granite"}}},
127                         "llm_factory": {"models": {"granite": {"model": "granite3.3:8b"}, "secure": "wrong-model"}}}}
128    r = MagicMock(); r.output = "ok"; r.model_alias = "secure"; r.provider = "ollama"; r.latency_ms = 1; r.token_usage = {}
129    events = []
130    with patch.object(mod, "ModelRouterFactory") as f, patch.object(mod, "emit_trace_event", events.append):
131        f.get_router.return_value.invoke.return_value = r
132        mod.llm_invoke(cfg, InferenceRequest(prompt="x", task_type="general"))
133    assert events[0]["model"] == "granite3.3:8b"

148def test_synthesizer_does_not_inherit_the_first_agents_role(monkeypatch):
149    from k9_aif_abb.k9_squad.base_squad import BaseSquad
150    from k9_aif_abb.k9_core.agent.base_agent import BaseAgent
151
152    class IntakeAgent(BaseAgent):
153        def execute(self, payload):
154            return {"output": "x"}
155
156    agent = IntakeAgent(config={"k9_env": "test", "role": "You are a claims intake clerk", "goal": "intake",
157                                "inference": {"x": 1}})
158    squad = BaseSquad("s", [agent])
159    synth = squad._get_synthesizer({"mode": "always"})
160    assert "role" not in synth.config and "goal" not in synth.config and synth.config.get("inference") == {"x": 1}
161    squad2 = BaseSquad("s2", [agent]); squad2.config = {"k9_env": "test", "inference": {"y": 2}}
162    assert squad2._get_synthesizer({"mode": "always"}).config.get("inference") == {"y": 2}
